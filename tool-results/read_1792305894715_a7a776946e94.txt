   150→    /**
   151→     * Primary static programmatic entry point: Youpi.pay(options)
   152→     */
   153→    static async pay(options = {}) {
   154→      const instance = new YoupiCheckout(options);
   155→
   156→      // If checkoutUrl is already provided, open modal directly
   157→      if (instance.options.checkoutUrl) {
   158→        instance.open();
   159→        return instance;
   160→      }
   161→
   162→      // If paymentId is already provided, construct checkoutUrl and open
   163→      if (instance.options.paymentId) {
   164→        instance.options.checkoutUrl = `${instance.options.portalUrl}/checkout/${instance.options.paymentId}`;
   165→        instance.open();
   166→        return instance;
   167→      }
   168→
   169→      // Otherwise, call YOUPI PG to create the payment session
   170→      try {
   171→        const orderId = instance.options.orderId || `ORDER-${Date.now()}`;
   172→        instance.options.orderId = orderId;
   173→        instance.options.order_id = orderId;
   174→
   175→        const customer = instance.options.customer || {};
   176→        const custPayload = {
   177→          fullName: customer.name || customer.fullName || '',
   178→          email: customer.email || '',
   179→          phoneNumber: customer.phone || customer.phoneNumber || customer.mobileNumber || ''
   180→        };
   181→
   182→        const createRes = await fetch(`${instance.options.apiUrl}/api/v1/payments`, {
   183→          method: 'POST',
   184→          headers: {
   185→            'Content-Type': 'application/json',
   186→            ...(instance.options.key ? { 'X-Merchant-Id': instance.options.key } : {})
   187→          },
   188→          body: JSON.stringify({
   189→            orderId: orderId,
   190→            amount: instance.options.amount,
   191→            currency: instance.options.currency || 'INR',
   192→            customer: (custPayload.email || custPayload.phoneNumber || custPayload.fullName) ? custPayload : undefined,
   193→            description: instance.options.description,
   194→            metadata: instance.options.metadata,
   195→            returnUrl: instance.options.successUrl || undefined
   196→          })
   197→        });
   198→
   199→        const data = await createRes.json();
   200→        if (!createRes.ok || !data.checkoutUrl) {
   201→          const errMsg = data.error || data.message || `Failed to create payment session (HTTP ${createRes.status})`;
   202→          const failResult = {
   203→            status: 'FAILED',
   204→            order_id: orderId,
   205→            reason: errMsg,
   206→            error: errMsg
   207→          };
   208→          YoupiCheckout._emitEvent('youpi:payment:failed', failResult);
   209→          YoupiCheckout._emitEvent('youpi:payment:result', failResult);
   210→          if (typeof instance.options.onFailed === 'function') instance.options.onFailed(failResult);
   211→          if (typeof instance.options.onPaymentFailed === 'function') instance.options.onPaymentFailed(failResult);
   212→          throw new Error(errMsg);
   213→        }
   214→
   215→        instance.options.paymentId = data.paymentId;
   216→        instance.options.checkoutUrl = data.checkoutUrl;
   217→        instance.open();
   218→        return instance;
   219→      } catch (err) {
   220→        console.error('[YouPI JS] Payment initialization error:', err.message);
   221→        throw err;
   222→      }
   223→    }

   237→    open() {
   238→      if (!isBrowser) {

   240→        return;

   247→      let targetUrl = '';
   248→      if (this.options.checkoutUrl) {
   249→        targetUrl = this.options.checkoutUrl;
   250→      } else if (this.options.paymentId) {
   251→        targetUrl = `${this.options.portalUrl}/checkout/${this.options.paymentId}`;
   252→      } else {
   253→        targetUrl = `${this.options.portalUrl}/checkout/demo`;
   254→      }
   255→
   256→      // If running on localhost / 127.0.0.1 and targetUrl points to remote domain, normalize to localhost host
   257→      if (isBrowser && (window.location.hostname === 'localhost' || window.location.hostname === '127.0.0.1')) {
   258→        if (targetUrl.includes('gauravtesting.online')) {
   259→          try {
   260→            const parsed = new URL(targetUrl);
   261→            targetUrl = `${window.location.protocol}//${window.location.host}${parsed.pathname}${parsed.search}`;
   262→          } catch {}
   263→        }
   264→      }
   265→
   266→      // Add embedded flag and custom params
   267→      const sep = targetUrl.includes('?') ? '&' : '?';
   268→      targetUrl = `${targetUrl}${sep}embedded=1`;
   269→
   270→      if (this.options.amount) {
   271→        targetUrl += `&customAmount=${encodeURIComponent(this.options.amount)}`;
   272→      }
   273→      if (this.options.name) {
   274→        targetUrl += `&customName=${encodeURIComponent(this.options.name)}`;
   275→      }
   276→
   277→      this._injectModal(targetUrl);
   278→      window.addEventListener('message', this._handleMessage);
   279→    }

   316→      const backdrop = document.createElement('div');

   330→            iframe.contentWindow.postMessage({ event: 'youpi.request_cancel' }, '*');

   332→          }
   333→        } catch (err) { }

   336→      };

   338→      const iframe = document.createElement('iframe');
   339→      iframe.className = 'youpi-checkout-iframe';
   340→      iframe.src = targetUrl;
   341→      iframe.allow = 'payment';
   342→
   343→      container.appendChild(closeBtn);
   344→      container.appendChild(iframe);
   345→      backdrop.appendChild(container);
   346→      document.body.appendChild(backdrop);
   347→
   348→      backdrop.classList.add('active');
   349→      container.classList.add('active');

   351→      this.backdrop = backdrop;
   352→      this.container = container;
   353→      this.iframe = iframe;
   354→
   355→      // Invariant: ONLY the cross (X) button exits the payment gateway!
   356→      // Prevent outside touch/click on backdrop from closing the checkout modal.
   357→      const stopOutsideDismiss = (e) => {
   358→        if (e.target === backdrop) {
   359→          e.preventDefault();
   360→          e.stopPropagation();
   361→          if (e.stopImmediatePropagation) e.stopImmediatePropagation();
   362→          return false;
   363→        }
   364→      };
   365→      backdrop.addEventListener('click', stopOutsideDismiss, true);
   366→      backdrop.addEventListener('touchstart', stopOutsideDismiss, { passive: false, capture: true });
   367→      backdrop.addEventListener('touchend', stopOutsideDismiss, { passive: false, capture: true });
   368→      backdrop.addEventListener('touchmove', stopOutsideDismiss, { passive: false, capture: true });
   369→      backdrop.addEventListener('pointerdown', stopOutsideDismiss, true);
   370→      backdrop.addEventListener('pointerup', stopOutsideDismiss, true);
   371→      backdrop.addEventListener('mousedown', stopOutsideDismiss, true);
   372→      backdrop.addEventListener('mouseup', stopOutsideDismiss, true);
   373→    }

   375→    _handleMessage(e) {
   376→      if (!e.data || typeof e.data !== 'object') return;
   377→
   378→      const event = e.data.event;
   379→      const rawData = e.data.data || e.data;
   380→
   381→      if (event === 'youpi.payment_success') {
   382→        const payload = e.data.data || {};
   383→        const result = {
   384→          status: 'SUCCESS',
   385→          order_id: payload.orderId || this.options.orderId || this.options.order_id,
   386→          payment_mode: payload.method || 'UPI',
   387→          amount_received: Number(payload.amount || this.options.amount || 0),
   388→          utr: payload.bankRef || payload.utr || payload.reference || '',
   389→          timestamp: payload.completedAt || new Date().toISOString(),
   390→          metadata: payload.metadata || this.options.metadata || {},
   391→          raw: payload
   392→        };
   393→
   394→        YoupiCheckout._emitEvent('youpi:payment:success', result);
   395→        YoupiCheckout._emitEvent('youpi:payment:result', result);
   396→
   397→        if (typeof this.options.handler === 'function') {
   398→          this.options.handler(result);
   399→        }
   400→        if (typeof this.options.onSuccess === 'function') {
   401→          this.options.onSuccess(result);

   404→        setTimeout(() => {

   406→            if (this.options.successUrl) {
   407→              const delim = this.options.successUrl.includes('?') ? '&' : '?';

   409→            }
