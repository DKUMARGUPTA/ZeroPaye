     5→/**
     6→ * Validates device access to an active payment session.
     7→ * Enforces:
     8→ * 1. If session has no device binding yet, binds to requesting device if provided.
     9→ * 2. If session is bound to Device A, Device B receives DUPLICATE_SESSION_DETECTED (HTTP 409).
    10→ */
    11→function checkSessionDeviceAccess(session, req, body = null, ctx = null) {
    12→  // If session is already in a terminal state, allow access so terminal status can be rendered
    13→  if ([PaymentStatus.SUCCESS, PaymentStatus.FAILED, PaymentStatus.EXPIRED, PaymentStatus.CANCELLED].includes(session.status)) {
    14→    return { allowed: true, boundDeviceId: session.deviceId, isTerminal: true };
    15→  }
    16→
    17→  // Internal server-side HTML render calls bypass device blocking
    18→  if (req.headers['x-internal-render'] === 'true') {
    19→    return { allowed: true, boundDeviceId: session.deviceId };
    20→  }
    21→
    22→  const requestingDeviceId = req.headers['x-device-id'] ||
    23→    body?.deviceId ||
    24→    (req.query && req.query.deviceId) ||
    25→    null;
    26→
    27→  // If session does not have a device binding yet:
    28→  if (!session.deviceId) {
    29→    if (requestingDeviceId) {
    30→      if (ctx?.deviceRepo) {
    31→        let dev = ctx.deviceRepo.findById(requestingDeviceId);
    32→        if (!dev) {
    33→          try {
    34→            ctx.deviceRepo.create({
    35→              id: requestingDeviceId,
    36→              publicKey: `adhoc_${requestingDeviceId}`,
    37→              fingerprintHash: `fp_${requestingDeviceId}`
    38→            });
    39→          } catch {}
    40→        } else {
    41→          ctx.deviceRepo.touch(requestingDeviceId);
    42→        }
    43→      }
    44→      if (ctx?.sessionRepo) {
    45→        try {
    46→          ctx.sessionRepo.bindDevice(session.id, requestingDeviceId);
    47→          session.deviceId = requestingDeviceId;
    48→        } catch (err) {
    49→          console.warn('[checkSessionDeviceAccess] bindDevice error:', err.message);
    50→        }
    51→      }
    52→    }
    53→    return { allowed: true, boundDeviceId: session.deviceId };
    54→  }
    55→
    56→  // If session HAS a bound device:
    57→  // Allow if requesting device matches bound device OR if requesting device is omitted on same session
    58→  if (!requestingDeviceId || requestingDeviceId === session.deviceId) {
    59→    if (requestingDeviceId && ctx?.deviceRepo) {
    60→      ctx.deviceRepo.touch(requestingDeviceId);
    61→    }
    62→    return { allowed: true, boundDeviceId: session.deviceId };
    63→  }
    64→
    65→  // Device mismatch: only reject if an explicitly DIFFERENT device ID was provided!
    66→  return {
    67→    allowed: false,
    68→    statusCode: 409,
    69→    body: {
    70→      error: 'DUPLICATE_SESSION_DETECTED',
    71→      session_id: session.id,
    72→      conflicting_session_id: session.id,
    73→      action: 'END_SESSION_AND_CREATE_NEW',
    74→      message: 'This payment session was created on another device. For security reasons, this session cannot be continued from this device.',
    75→      boundDeviceId: session.deviceId,
    76→      requestingDeviceId
    77→    }
    78→  };
    79→}

    92→/**
    93→ * Determines whether UPI pollers / observers (Android SMS, Google Messages, Gmail) are active.
    94→ * Strict Rule: Healthy (WORKING) ONLY IF:
    95→ * 1. Gmail OAuth is configured AND Gmail observer is actively polling.
    96→ * 2. SMS Gateway is connected (Android app or Google Messages) AND actively polling / reflecting messages.
    97→ * If either is missing, unconfigured, or offline, status is ISSUES with downtime warning.
    98→ */
    99→async function checkUpiServerHealth(ctx, enabledMethods = [], sellingMode = 'OFFLINE', merchantId = '00000000-0000-0000-0000-000000000001') {
   100→  // 0. Check False Narrative Dev Switch (Active for 10 minutes from enabling time)
   101→  try {
   102→    const merchant = ctx.merchantRepo?.findById ? ctx.merchantRepo.findById(merchantId) : null;
   103→    const falseNarrativeUntil = merchant?.branding?.falseNarrativeUntil;
   104→    const falseNarrativeMode = (merchant?.branding?.falseNarrativeMode || 'GREEN').toUpperCase();
   105→    if (falseNarrativeUntil && Number(falseNarrativeUntil) > Date.now()) {
   106→      const remainingSeconds = Math.max(0, Math.round((Number(falseNarrativeUntil) - Date.now()) / 1000));
   107→      if (falseNarrativeMode === 'GREEN') {
   108→        return {
   109→          status: 'WORKING',
   110→          tier: 'GREEN',
   111→          label: 'Server Is Working',
   112→          isWorking: true,
   113→          reason: 'False Narrative Active (Dev Override: GREEN): UPI servers interpreted as operational',
   114→          smsHealthy: true,
   115→          gmailHealthy: true,
   116→          warning: null,
   117→          recommendedAlternative: null,
   118→          isFalseNarrative: true,
   119→          narrative: 'GREEN',
   120→          falseNarrativeRemainingSeconds: remainingSeconds
   121→        };
   122→      } else if (falseNarrativeMode === 'YELLOW') {
   123→        return {
   124→          status: 'BUSY',
   125→          tier: 'YELLOW',
   126→          label: 'UPI Server Busy',
   127→          isWorking: true,
   128→          reason: 'False Narrative Active (Dev Override: YELLOW): UPI servers experiencing processing delays',
   129→          smsHealthy: true,
   130→          gmailHealthy: false,
   131→          warning: 'Your payment may go through extra processing seconds.',
   132→          recommendedAlternative: null,
   133→          isFalseNarrative: true,
   134→          narrative: 'YELLOW',
   135→          falseNarrativeRemainingSeconds: remainingSeconds
   136→        };
   137→      } else if (falseNarrativeMode === 'RED') {
   138→        return {
   139→          status: 'ISSUES',
   140→          tier: 'RED',
   141→          label: 'Server Facing Issues',
   142→          isWorking: false,
   143→          reason: 'False Narrative Active (Dev Override: RED): UPI servers facing downtime',
   144→          smsHealthy: false,
   145→          gmailHealthy: false,
   146→          warning: 'It is not recommended to pay by UPI right now as the server is facing downtime.',
   147→          recommendedAlternative: null,
   148→          isFalseNarrative: true,
   149→          narrative: 'RED',
   150→          falseNarrativeRemainingSeconds: remainingSeconds
   151→        };
   152→      }
   153→    }
   154→  } catch (err) {}
   155→
   156→  let smsHealthy = false;
   157→  let gmailHealthy = false;
   158→  let smsReason = 'SMS gateway disconnected';
   159→  let gmailReason = 'Gmail OAuth unconfigured';
   160→
   161→  // 1. Check SMS Gateway (Android app or Google Messages)
   162→  try {
   163→    if (ctx.db) {
   164→      const activeAndroid = ctx.db.queryOne(`
   165→        SELECT id, device_name, last_heartbeat_at
   166→        FROM android_devices
   167→        WHERE is_active = 1 AND last_heartbeat_at >= datetime('now', '-5 minutes')
   168→        LIMIT 1
   169→      `);
   170→      if (activeAndroid) {
   171→        smsHealthy = true;
   172→        smsReason = `Android SMS Gateway Active (${activeAndroid.device_name || 'Enrolled Device'})`;
   173→      }
   174→    }
   175→  } catch (err) {}
   176→
   177→  if (!smsHealthy) {
   178→    try {
   179→      const smsPort = Number(process.env.SMS_OBSERVER_PORT || 6002);
   180→      const res = await fetch(`http://127.0.0.1:${smsPort}/api/browser/status`, { signal: AbortSignal.timeout(800) });
   181→      if (res.ok) {
   182→        const data = await res.json();
   183→        const isRecent = data.lastSeenAt && (Date.now() - new Date(data.lastSeenAt).getTime() < 300000);
   184→        if (data.isPaired === true && data.status === 'CONNECTED' && data.mode === 'BACKGROUND' && (data.isPolling === true || isRecent)) {
   185→          smsHealthy = true;
   186→          smsReason = 'Google Messages SMS Gateway active & paired';
   187→        }
   188→      }
   189→    } catch (err) {}
   190→  }
   191→
   192→  // 2. Check Gmail OAuth Configuration & Active Poller
   193→  try {
   194→    const merchant = ctx.merchantRepo?.findById ? ctx.merchantRepo.findById(merchantId) : null;
   195→    const gmailConfig = merchant?.branding?.gmailConfig || {};
   196→    const hasRefreshToken = Boolean(gmailConfig.refreshToken && gmailConfig.refreshToken.trim());
   197→    const hasApiKey = Boolean(gmailConfig.apiKey && gmailConfig.apiKey.trim());
   198→
   199→    if (hasRefreshToken || hasApiKey) {
   200→      let pollerActive = false;
   201→      const gmailPort = Number(process.env.GMAIL_OBSERVER_PORT || 6003);
   202→
   203→      try {
   204→        const res = await fetch(`http://127.0.0.1:${gmailPort}/health`, { signal: AbortSignal.timeout(800) });
   205→        if (res.ok) {
   206→          const data = await res.json();
   207→          if (data.status === 'UP') {
   208→            pollerActive = true;
   209→          }
   210→        }
   211→      } catch (e) {}
   212→
   213→      if (!pollerActive) {
   214→        try {
   215→          const res = await fetch(`http://127.0.0.1:${gmailPort}/api/browser/status`, { signal: AbortSignal.timeout(800) });
   216→          if (res.ok) {
   217→            const data = await res.json();
   218→            if (data.isLoggedIn || data.status === 'CONNECTED' || data.isPolling) {
   219→              pollerActive = true;
   220→            }
   221→          }
   222→        } catch (e) {}
   223→      }
   224→
   225→      if (!pollerActive) {
   226→        try {
   227→          const res = await fetch(`http://127.0.0.1:6099/health`, { signal: AbortSignal.timeout(800) });
   228→          if (res.ok) {
   229→            const data = await res.json();
   230→            if (data.status === 'UP' && (data.isPolling || data.pollerActive !== false)) {
   231→              pollerActive = true;
   232→            }
   233→          }
   234→        } catch (e) {}
   235→      }
   236→
   237→      if (pollerActive || hasRefreshToken) {
   238→        gmailHealthy = true;
   239→        gmailReason = 'Gmail OAuth connected and actively polling';
   240→      } else {
   241→        gmailReason = 'Gmail OAuth configured but observer service is offline';
   242→      }
   243→    } else {
   244→      // Check if browser-based Gmail automator is actively logged in
   245→      try {
   246→        const gmailPort = Number(process.env.GMAIL_OBSERVER_PORT || 6003);
   247→        const res = await fetch(`http://127.0.0.1:${gmailPort}/api/browser/status`, { signal: AbortSignal.timeout(800) });
   248→        if (res.ok) {
   249→          const data = await res.json();
   250→          if (data.isLoggedIn === true && data.status === 'CONNECTED') {
   251→            gmailHealthy = true;
   252→            gmailReason = 'Gmail browser automator logged in';
   253→          }
   254→        }
   255→      } catch (e) {}
   256→
   257→      if (!gmailHealthy) {
   258→        gmailReason = 'Gmail OAuth not configured';
   259→      }
   260→    }
   261→  } catch (err) {
   262→    gmailReason = `Gmail check error: ${err.message}`;
   263→  }
   264→
   265→  // 3-Tier Health Model:
   266→  // Green: Both SMS + Gmail connected -> healthy
   267→  // Yellow: Either SMS OR Gmail connected -> extra processing time
   268→  // Red: Neither connected -> server facing issues
   269→  const hasCash = sellingMode === 'ONLINE'
   270→    ? enabledMethods.includes('COD')
   271→    : enabledMethods.includes('CASH_COUNTER');
   272→
   273→  if (smsHealthy && gmailHealthy) {
   274→    return {
   275→      status: 'WORKING',
   276→      tier: 'GREEN',
   277→      label: 'Server Is Working',
   278→      isWorking: true,
   279→      reason: 'All observers active: Gmail OAuth polling & SMS Gateway connected',
   280→      smsHealthy: true,
   281→      gmailHealthy: true,
   282→      warning: null,
   283→      recommendedAlternative: null
   284→    };
   285→  }
   286→
   287→  if (smsHealthy || gmailHealthy) {
   288→    return {
   289→      status: 'BUSY',
   290→      tier: 'YELLOW',
   291→      label: 'UPI Server Busy',
   292→      isWorking: true,
   293→      reason: smsHealthy ? 'SMS observer active, Gmail observer offline' : 'Gmail observer active, SMS observer offline',
   294→      smsHealthy,
   295→      gmailHealthy,
   296→      warning: 'Your payment may go through extra processing seconds.',
   297→      recommendedAlternative: null
   298→    };
   299→  }
   300→
   301→  return {
   302→    status: 'ISSUES',
   303→    tier: 'RED',
   304→    label: 'Server Facing Issues',
   305→    isWorking: false,
   306→    reason: 'Both Gmail OAuth and SMS Gateway are offline or unconfigured',
   307→    smsHealthy: false,
   308→    gmailHealthy: false,
   309→    warning: 'It is not recommended to pay by UPI right now as the server is facing downtime. We recommend paying by cash instead.',
   310→    recommendedAlternative: hasCash ? 'CASH' : null
   311→  };
   312→}

   314→export function registerCheckoutRoutes(server, ctx) {
   315→  // GET /api/v1/checkout/:id/state
   316→  server.get('/api/v1/checkout/:id/state', async (req, res, params) => {
   317→    const session = ctx.sessionRepo.findById(params.id);
   318→    if (!session) {
   319→      HttpServer.json(res, 404, { error: `Checkout session not found: ${params.id}` });
   320→      return;
   321→    }
   322→
   323→    // Server-side Ban & Fraud Enforcement
   324→    const isCustomerBanned = session.customerId && ctx.customerRepo && ctx.customerRepo.isBanned(session.customerId);
   325→    const isSessionBlocked = session.status === 'BLOCKED' || (session.metadata?.banReason === 'FRAUDULENT_PAYMENT_INJECTION');
   326→
   327→    if (isCustomerBanned || isSessionBlocked) {
   328→      const cust = session.customerId ? ctx.customerRepo.findById(session.customerId) : null;
   329→      HttpServer.json(res, 200, {
   330→        id: session.id,
   331→        orderId: session.orderId,
   332→        amount: toRupeesString(session.amountPaise),
   333→        status: 'BLOCKED',
   334→        isBanned: true,
   335→        fraudDetected: true,
   336→        customerName: cust?.name || 'Customer',
   337→        blockedNotice: {
   338→          title: 'FRAUDULENT ACTIVITY DETECTED',
   339→          subtitle: 'Payment Session Terminated & Account Suspended',
   340→          message: 'Our automated verification engine detected fraudulent payment activity associated with this account. Access has been immediately suspended while our security operations team investigates the issue.',
   341→          investigating: true,
   342→          reason: cust?.banReason || 'FRAUDULENT_PAYMENT_INJECTION'
   343→        }
   344→      });
   345→      return;
   346→    }
   347→
   348→    // Authoritative Server-Side Expiration Enforcer
   349→    const now = Date.now();
   350→    if (session.status === PaymentStatus.ACTIVE && session.expiresAt && session.expiresAt.getTime() <= now) {
   351→      ctx.sessionRepo.transitionToTerminal(session.id, PaymentStatus.EXPIRED);
   352→      session.status = PaymentStatus.EXPIRED;
   353→      if (session.receivingAccountId) {
   354→        ctx.queueEngine.promoteNext(session.receivingAccountId);
   355→      }
   356→      server.broadcastSse(session.id, 'status_change', { status: PaymentStatus.EXPIRED });
   357→    } else if (session.status === PaymentStatus.PROCESSING && session.processingExpiresAt && session.processingExpiresAt.getTime() <= now) {
   358→      ctx.sessionRepo.transitionToTerminal(session.id, PaymentStatus.FAILED);
   359→      session.status = PaymentStatus.FAILED;
   360→      if (session.receivingAccountId) {
   361→        ctx.queueEngine.promoteNext(session.receivingAccountId);
   362→      }
   363→      server.broadcastSse(session.id, 'status_change', { status: PaymentStatus.FAILED });
   364→    }
   365→
   366→    // Terminal session handling: if session is terminal, return terminal state directly (CASE 4)
   367→    if ([PaymentStatus.SUCCESS, PaymentStatus.FAILED, PaymentStatus.EXPIRED, PaymentStatus.CANCELLED].includes(session.status)) {
   368→      const merchant = ctx.merchantRepo.findById(session.merchantId);
   369→      const branding = {
   370→        appName: 'YOUPI PG',
   371→        primaryColor: '#2563eb',
   372→        ...(merchant?.branding || {})
   373→      };
   374→      HttpServer.json(res, 200, {
   375→        id: session.id,
   376→        paymentId: session.id,
   377→        orderId: session.orderId,
   378→        amount: toRupeesString(session.amountPaise),
   379→        amountPaise: session.amountPaise.toString(),
   380→        currency: session.currency,
   381→        status: session.status,
   382→        timeRemainingSeconds: 0,
   383→        isTerminal: true,
   384→        branding
   385→      });
   386→      return;
   387→    }
   388→
   389→    // Device-level session binding validation (Section 8 & 9)
   390→    const deviceAccess = checkSessionDeviceAccess(session, req, null, ctx);
   391→    if (!deviceAccess.allowed) {
   392→      HttpServer.json(res, deviceAccess.statusCode || 409, deviceAccess.body);
   393→      return;
   394→    }
   395→
   396→    const merchant = ctx.merchantRepo.findById(session.merchantId);
   397→    const snapshot = session.metadata?.receivingAccountSnapshot;
   398→    let receivingAccount = session.receivingAccountId
   399→      ? ctx.accountRepo.findById(session.receivingAccountId)
   400→      : null;
   401→
   402→    if (!receivingAccount && snapshot?.id) {
   403→      receivingAccount = ctx.accountRepo.findById(snapshot.id);
   404→    }
   405→    if (!receivingAccount && session.method === 'UPI') {
   406→      receivingAccount = ctx.accountRepo.findActiveForMerchant(session.merchantId);
   407→    }
   408→
   409→    const payeeVpa = receivingAccount?.upiId || snapshot?.upiId || null;
   410→    const payeeName = receivingAccount?.payeeName || snapshot?.payeeName || merchant?.businessName || 'Merchant Store';
   411→    const isMerchantEnabled = receivingAccount
   412→      ? Boolean(receivingAccount.isMerchantEnabled)
   413→      : (snapshot?.isMerchantEnabled !== undefined ? Boolean(snapshot.isMerchantEnabled) : true);
   414→    const upiMode = isMerchantEnabled ? 'MERCHANT' : 'NON_MERCHANT';
   415→
   416→    let timeRemainingSeconds = 0;
   417→    if (session.status === PaymentStatus.ACTIVE && session.expiresAt) {
   418→      timeRemainingSeconds = Math.max(0, Math.ceil((session.expiresAt.getTime() - now) / 1000));
   419→    } else if (session.status === PaymentStatus.PROCESSING && session.processingExpiresAt) {
   420→      timeRemainingSeconds = Math.max(0, Math.ceil((session.processingExpiresAt.getTime() - now) / 1000));
   421→    } else if (session.status === PaymentStatus.QUEUED) {
   422→      timeRemainingSeconds = ctx.queueEngine.calculateEstimatedWaitSeconds(
   423→        session.receivingAccountId,
   424→        session.queuePosition
   425→      );
   426→    }
   427→
   428→    let upiUri = null;
   429→    if (session.status === PaymentStatus.ACTIVE && payeeVpa) {
   430→      try {
   431→        upiUri = buildUpiUri({
   432→          payeeVpa,
   433→          payeeName,
   434→          amountPaise: session.amountPaise,
   435→          transactionRef: session.orderId,
   436→          currency: session.currency || 'INR'
   437→        });
   438→      } catch {
   439→        upiUri = null;
   440→      }
   441→    }
   442→
   443→    const branding = {
   444→      appName: 'YOUPI PG',
   445→      primaryColor: '#2563eb',
   446→      headerTheme: 'gradient',
   447→      logoUrl: null,
   448→      counterInstructions: 'Please visit the billing counter and present this screen to the cashier.',
   449→      cardCounterInstructions: 'Please swipe, insert, or tap your credit or debit card on the store POS terminal at the billing desk.',
   450→      ...(merchant?.branding || {})
   451→    };
   452→
   453→    const supportEmails = branding.supportEmails || (merchant?.contactEmail ? [merchant.contactEmail] : []);
   454→    const supportMobiles = branding.supportMobiles || (merchant?.contactPhone ? [merchant.contactPhone] : []);
   455→    const privacyPolicyUrl = branding.privacyPolicyUrl || null;
   456→    branding.supportEmails = supportEmails;
   457→    branding.supportMobiles = supportMobiles;
   458→    const isCounterSession = session.method === 'CASH_COUNTER' || session.method === 'CARD_COUNTER';
   459→    const merchantBranding = merchant?.branding || {};
   460→    const sellingMode = merchantBranding.sellingMode || (isCounterSession ? 'OFFLINE' : 'ONLINE');
   461→    branding.sellingMode = sellingMode;
   462→
   463→    let rawEnabled = merchantBranding.enabledPaymentMethods || (sellingMode === 'ONLINE' ? ['UPI', 'COD'] : ['UPI', 'CASH_COUNTER', 'CARD_COUNTER']);
   464→    let enabledMethods = ['UPI'];
   465→    if (sellingMode === 'ONLINE') {
   466→      if (rawEnabled.includes('COD')) {
   467→        enabledMethods.push('COD');
   468→      }
   469→    } else {
   470→      if (rawEnabled.includes('CASH_COUNTER')) {
   471→        enabledMethods.push('CASH_COUNTER');
   472→      }
   473→      if (rawEnabled.includes('CARD_COUNTER')) {
   474→        enabledMethods.push('CARD_COUNTER');
   475→      }
   476→    }
   477→
   478→    const authHeader = req.headers['authorization'] || '';
   479→    const bearerToken = authHeader.startsWith('Bearer ') ? authHeader.slice(7).trim() : (req.headers['x-customer-token'] || null);
   480→
   481→    let customer = null;
   482→    if (session.customerId && ctx.customerRepo) {
   483→      const c = ctx.customerRepo.findById(session.customerId);
   484→      if (c) {
   485→        customer = {
   486→          id: c.id,
   487→          name: c.name,
   488→          email: c.email,
   489→          mobileNumber: c.mobileNumber,
   490→          fullName: c.name,
   491→          phoneNumber: c.mobileNumber
   492→        };
   493→      }
   494→    } else if (!customer && bearerToken && ctx.customerSessionRepo) {
   495→      const authContext = ctx.customerSessionRepo.findByToken(bearerToken);
   496→      if (authContext?.customer) {
   497→        const c = authContext.customer;
   498→        customer = {
   499→          id: c.id,
   500→          name: c.name,
   501→          email: c.email,
   502→          mobileNumber: c.mobileNumber,
   503→          fullName: c.name,
   504→          phoneNumber: c.mobileNumber
   505→        };
   506→      }
   507→    }
   508→
   509→    // Single active payment session check (Section 1 & 2)
   510→    const requestingDeviceId = req.headers['x-device-id'] || session.deviceId || null;
   511→    let hasActiveSessionConflict = false;
   512→    let conflictingSession = null;
   513→    const conflictingActive = ctx.sessionRepo.findConflictingActiveSession
   514→      ? ctx.sessionRepo.findConflictingActiveSession({
   515→          customerId: session.customerId,
   516→          deviceId: requestingDeviceId,
   517→          excludeSessionId: session.id
   518→        })
   519→      : null;
   520→
   521→    if (conflictingActive && conflictingActive.id !== session.id && [PaymentStatus.INITIATED, PaymentStatus.ACTIVE, PaymentStatus.QUEUED].includes(session.status)) {
   522→      if (requestingDeviceId && conflictingActive.deviceId && requestingDeviceId !== conflictingActive.deviceId) {
   523→        // CASE 3: Active session on another device -> DUPLICATE_SESSION_DETECTED

  1474→  // Cancels previous session created on another device and creates a fresh payment session bound to current device
  1475→  async function handleEndAndRestartPayment(req, res, params) {
  1476→    const session = ctx.sessionRepo.findById(params.id);
  1477→    const body = await HttpServer.parseBody(req).catch(() => ({}));
  1478→    const conflictingSessionId = body?.conflictingSessionId || body?.conflicting_session_id || null;
  1479→

  1481→      HttpServer.json(res, 404, { error: `Payment session not found: ${params.id}` });
  1482→      return;
  1483→    }
  1484→
  1485→    const targetSession = session || ctx.sessionRepo.findById(conflictingSessionId);
  1486→    if (!targetSession) {
  1487→      HttpServer.json(res, 404, { error: `Payment session not found: ${params.id}` });
  1488→      return;
  1489→    }
  1490→
  1491→    const newDeviceId = req.headers['x-device-id'] || body?.deviceId || null;
  1492→    if (!newDeviceId) {
  1493→      HttpServer.json(res, 400, { error: 'Current device identity is required to bind the new payment session.' });
  1494→      return;
  1495→    }
  1496→
  1497→    const sourceSession = (conflictingSessionId ? ctx.sessionRepo.findById(conflictingSessionId) : null) || targetSession;
  1498→    const customerId = body?.customerId || sourceSession?.customerId || targetSession.customerId || null;
  1499→
  1500→    // 1. Identify ALL sessions on old/other devices that must be cancelled
  1501→    const sessionsToTerminate = new Map();
  1502→
  1503→    if (![PaymentStatus.CANCELLED, PaymentStatus.EXPIRED, PaymentStatus.FAILED, PaymentStatus.SUCCESS].includes(targetSession.status)) {
  1504→      sessionsToTerminate.set(targetSession.id, targetSession);
  1505→    }
  1506→
  1507→    if (conflictingSessionId) {
  1508→      const confSess = ctx.sessionRepo.findById(conflictingSessionId);
  1509→      if (confSess && ![PaymentStatus.CANCELLED, PaymentStatus.EXPIRED, PaymentStatus.FAILED, PaymentStatus.SUCCESS].includes(confSess.status)) {
  1510→        sessionsToTerminate.set(confSess.id, confSess);
  1511→      }
  1512→    }
  1513→
  1514→    if (customerId && ctx.sessionRepo.findAllActiveOrPendingForCustomer) {
  1515→      const custSessions = ctx.sessionRepo.findAllActiveOrPendingForCustomer(customerId);
  1516→      for (const cs of custSessions) {
  1517→        if (![PaymentStatus.CANCELLED, PaymentStatus.EXPIRED, PaymentStatus.FAILED, PaymentStatus.SUCCESS].includes(cs.status)) {
  1518→          if (!cs.deviceId || cs.deviceId !== newDeviceId || cs.id === targetSession.id) {
  1519→            sessionsToTerminate.set(cs.id, cs);
  1520→          }
  1521→        }
  1522→      }
  1523→    }
  1524→
  1525→    if (session && session.id !== targetSession.id && ![PaymentStatus.CANCELLED, PaymentStatus.EXPIRED, PaymentStatus.FAILED, PaymentStatus.SUCCESS].includes(session.status)) {
  1526→      sessionsToTerminate.set(session.id, session);
  1527→    }
  1528→
  1529→    // Also check if an active session on the receiving account belongs to the same order / customer
  1530→    const receivingAccountIdForLookup = targetSession.receivingAccountId || session?.receivingAccountId;
  1531→    if (receivingAccountIdForLookup && ctx.sessionRepo.findCurrentActiveOrProcessing) {
  1532→      const activeOnAcc = ctx.sessionRepo.findCurrentActiveOrProcessing(receivingAccountIdForLookup);
  1533→      if (activeOnAcc && activeOnAcc.deviceId !== newDeviceId) {
  1534→        const cleanTargetOrder = targetSession.orderId ? targetSession.orderId.replace(/-R\d+$/, '') : '';
  1535→        const cleanActiveOrder = activeOnAcc.orderId ? activeOnAcc.orderId.replace(/-R\d+$/, '') : '';
  1536→        if ((cleanTargetOrder && cleanTargetOrder === cleanActiveOrder) || (customerId && activeOnAcc.customerId === customerId)) {
  1537→          sessionsToTerminate.set(activeOnAcc.id, activeOnAcc);
  1538→        }
  1539→      }
  1540→    }
  1541→
  1542→    // Resolve receiving account ID for new session (guarantee active account for merchant)
  1543→    const activeMerchantAcc = ctx.accountRepo ? ctx.accountRepo.findActiveForMerchant(sourceSession.merchantId) : null;
  1544→    const targetReceivingAccountId = activeMerchantAcc?.id || sourceSession.receivingAccountId;
  1545→
  1546→    const affectedAccountIds = new Set();
  1547→    for (const [sId, sObj] of sessionsToTerminate.entries()) {
  1548→      if (sObj?.receivingAccountId) {
  1549→        affectedAccountIds.add(sObj.receivingAccountId);
  1550→      }
  1551→    }
  1552→    if (receivingAccountIdForLookup) {
  1553→      affectedAccountIds.add(receivingAccountIdForLookup);
  1554→    }
  1555→    if (targetReceivingAccountId) {
  1556→      affectedAccountIds.add(targetReceivingAccountId);
  1557→    }
  1558→
  1559→    // 2. Terminate all old sessions, cancel timers, and broadcast SSE
  1560→    for (const [sId, sObj] of sessionsToTerminate.entries()) {
  1561→      ctx.sessionRepo.transitionToTerminal(sId, PaymentStatus.CANCELLED, {
  1562→        reason: 'REPLACED_ON_ANOTHER_DEVICE',
  1563→        replacedByDeviceId: newDeviceId
  1564→      });
  1565→
  1566→      if (ctx.timingEngine) {
  1567→        ctx.timingEngine.cancelActiveTimer(sId);
  1568→        if (typeof ctx.timingEngine.clearTimers === 'function') {
  1569→          ctx.timingEngine.clearTimers(sId);
  1570→        }
  1571→      }
  1572→
  1573→      server.broadcastSse(sId, 'status_change', {
  1574→        status: PaymentStatus.CANCELLED,
  1575→        reason: 'REPLACED_ON_ANOTHER_DEVICE',
  1576→        message: 'Payment session closed because you continued on another device.'
  1577→      });
  1578→    }
  1579→
  1580→    // 3. Immediately release receiving-account slot, re-index queue, and promote next queued session if applicable
  1581→    const termIds = Array.from(sessionsToTerminate.keys());
  1582→    for (const accId of affectedAccountIds) {
  1583→      const nowIso = new Date().toISOString();
  1584→      if (termIds.length > 0) {
  1585→        ctx.db.execute(`
  1586→          UPDATE payment_receiving_accounts
  1587→          SET is_slot_busy = 0, current_active_session_id = NULL, updated_at = ?
  1588→          WHERE id = ? AND (current_active_session_id IN (${termIds.map(() => '?').join(',')}) OR current_active_session_id IS NULL)
  1589→        `, [nowIso, accId, ...termIds]);
  1590→      }
  1591→
  1592→      if (ctx.sessionRepo.reindexQueue) {
  1593→        ctx.sessionRepo.reindexQueue(accId);
  1594→      }
  1595→
  1596→      if (ctx.queueEngine && typeof ctx.queueEngine.promoteNext === 'function') {
  1597→        const promoted = ctx.queueEngine.promoteNext(accId);
  1598→        if (promoted) {
  1599→          if (ctx.timingEngine && promoted.expiresAt) {
  1600→            ctx.timingEngine.startActiveTimer(promoted.id, promoted.expiresAt);
  1601→          }
  1602→          server.broadcastSse(promoted.id, 'status_change', {
  1603→            status: PaymentStatus.ACTIVE,
  1604→            queuePosition: 0,
  1605→            expiresAt: promoted.expiresAt
  1606→          });
  1607→          server.broadcastSse('admin_counter_events', 'status_change', {
  1608→            status: PaymentStatus.ACTIVE,
  1609→            sessionId: promoted.id,
  1610→            orderId: promoted.orderId,
  1611→            amountPaise: promoted.amountPaise ? promoted.amountPaise.toString() : '0'
  1612→          });
  1613→        }
  1614→      }
  1615→    }
  1616→
  1617→    // 4. Generate unique order reference from sourceSession
  1618→    const cleanOrderId = sourceSession.orderId ? sourceSession.orderId.replace(/-R\d+$/, '') : `ORD-${Date.now()}`;
  1619→    const randSuffix = Math.floor(1000 + Math.random() * 9000);
  1620→    const newOrderId = `${cleanOrderId}-R${randSuffix}`;
  1621→
  1622→    // 5. Ensure device exists and register
  1623→    if (newDeviceId && ctx.deviceRepo) {
  1624→      let dev = ctx.deviceRepo.findById(newDeviceId);
  1625→      if (!dev) {
  1626→        try {
  1627→          ctx.deviceRepo.create({
  1628→            id: newDeviceId,
  1629→            publicKey: `adhoc_${newDeviceId}`,
  1630→            fingerprintHash: `fp_${newDeviceId}`
  1631→          });
  1632→        } catch {}
  1633→      } else {
  1634→        ctx.deviceRepo.touch(newDeviceId);
  1635→      }
  1636→    }
  1637→
  1638→    // 6. Create fresh payment session bound to newDeviceId
  1639→    const newSession = ctx.sessionRepo.create({
  1640→      merchantId: sourceSession.merchantId,
  1641→      customerId: customerId || sourceSession.customerId,
  1642→      receivingAccountId: targetReceivingAccountId,
  1643→      orderId: newOrderId,
  1644→      amountPaise: sourceSession.amountPaise,
  1645→      currency: sourceSession.currency,
  1646→      method: sourceSession.method || PaymentMethod.UPI,
  1647→      status: PaymentStatus.INITIATED,
  1648→      returnUrl: sourceSession.returnUrl,
  1649→      deviceId: newDeviceId,
  1650→      metadata: {
  1651→        ...(sourceSession.metadata || {}),
  1652→        reboundFromSessionId: sourceSession.id,
  1653→        reboundAt: new Date().toISOString()
  1654→      }
  1655→    });
  1656→
  1657→    if (customerId && ctx.deviceRepo) {
  1658→      try { ctx.deviceRepo.associateCustomer(newDeviceId, customerId); } catch {}
  1659→    }
  1660→
  1661→    // 7. Enqueue or activate new session if UPI
  1662→    let queueResult = { status: newSession.status, queuePosition: 0, expiresAt: null };
  1663→    if (newSession.receivingAccountId && (String(newSession.method).toUpperCase() === 'UPI' || !newSession.method)) {
  1664→      queueResult = ctx.queueEngine.enqueueOrActivate(newSession.id, newSession.receivingAccountId);
  1665→      if (queueResult.expiresAt && ctx.timingEngine) {
  1666→        ctx.timingEngine.startActiveTimer(newSession.id, queueResult.expiresAt);
  1667→      }
  1668→      server.broadcastSse(newSession.id, 'status_change', {
  1669→        status: queueResult.status,
  1670→        queuePosition: queueResult.queuePosition,
  1671→        expiresAt: queueResult.expiresAt
  1672→      });
  1673→      server.broadcastSse('admin_counter_events', 'status_change', {
  1674→        status: queueResult.status,
  1675→        sessionId: newSession.id,
  1676→        orderId: newSession.orderId,
  1677→        amountPaise: newSession.amountPaise ? newSession.amountPaise.toString() : '0'
  1678→      });
  1679→    }
  1680→
  1681→    if (ctx.auditRepo) {
  1682→      ctx.auditRepo.log({
  1683→        merchantId: sourceSession.merchantId,
  1684→        entityType: 'PAYMENT_SESSION',
  1685→        entityId: newSession.id,
  1686→        action: 'PAYMENT_SESSION_RECREATED_ON_NEW_DEVICE',
  1687→        actorType: 'CUSTOMER',
  1688→        actorId: customerId || 'ANONYMOUS',
  1689→        details: {
  1690→          terminatedSessionIds: Array.from(sessionsToTerminate.keys()),
  1691→          newSessionId: newSession.id,
  1692→          newDeviceId
  1693→        }
  1694→      });
  1695→    }
  1696→
  1697→    HttpServer.json(res, 201, {
  1698→      success: true,
  1699→      oldSessionId: targetSession.id,
  1700→      terminatedSessionIds: Array.from(sessionsToTerminate.keys()),
  1701→      newSessionId: newSession.id,
  1702→      paymentId: newSession.id,
  1703→      orderId: newSession.orderId,
  1704→      checkoutUrl: `/checkout/${newSession.id}`,
  1705→      status: queueResult.status,
  1706→      queuePosition: queueResult.queuePosition,
  1707→      expiresAt: queueResult.expiresAt,
  1708→      message: 'Previous sessions terminated. New payment session activated and bound to this device.'
  1709→    });
  1710→  }
  1711→
  1712→  server.post('/api/v1/checkout/:id/end-and-restart-payment', handleEndAndRestartPayment);
  1713→  server.post('/api/v1/checkout/:id/end-session-and-create-new', handleEndAndRestartPayment);
  1714→
  1715→  // GET /api/v1/checkout/:id/stream and /events (SSE)
  1716→  server.get('/api/v1/checkout/:id/stream', async (req, res, params) => {
  1717→    const session = ctx.sessionRepo.findById(params.id);
  1718→    if (!session) {
  1719→      HttpServer.json(res, 404, { error: 'Session not found' });
