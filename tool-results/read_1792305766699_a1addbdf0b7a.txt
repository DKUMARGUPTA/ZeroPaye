    57→    57→/**
    58→    58→ * Ensures any pairing string or QR data is formatted as an official WhatsApp Web
    59→    59→ * pairing link (https://wa.me/settings/linked_devices#2@...) and converted to an
    60→    60→ * authentic, renderable Data URI (PNG or SVG).
    61→    61→ */
    62→    62→export async function formatQrDataUrl(rawText, qrcodeModule = null) {
    63→    63→  if (!rawText) return null;
    64→    64→  if (typeof rawText === 'string' && rawText.startsWith('data:image/')) {
    65→    65→    return rawText;
    66→    66→  }
    67→    67→
    68→    68→  let text = typeof rawText === 'string' ? rawText : String(rawText);
    69→    69→  if (text.startsWith('2@') || text.startsWith('1@')) {
    70→    70→    text = `https://wa.me/settings/linked_devices#${text}`;
    71→    71→  }
    72→    72→
    73→    73→  let qrEngine = qrcodeModule || qrcodeLib;
    74→    74→  if (!qrEngine) {
    75→    75→    const deps = await loadDependencies();
    76→    76→    qrEngine = deps.qrcode;
    77→    77→  }
    78→    78→  if (qrEngine && typeof qrEngine.toDataURL === 'function') {
    79→    79→    try {
    80→    80→      const dataUrl = await qrEngine.toDataURL(text, {
    81→    81→        width: 320,
    82→    82→        margin: 2,
    83→    83→        errorCorrectionLevel: 'M',
    84→    84→        color: { dark: '#0f172a', light: '#ffffff' }
    85→    85→      });
    86→    86→      if (dataUrl && dataUrl.startsWith('data:image/')) {
    87→    87→        return dataUrl;
    88→    88→      }
    89→    89→    } catch (e) {
    90→    90→      console.warn('[WhatsAppBerryService] qrcode.toDataURL notice:', e.message);
    91→    91→    }
    92→    92→  }
    93→    93→  try {
    94→    94→    const svg = generateQrSvg(text, 320, 'M');
    95→    95→    return 'data:image/svg+xml;base64,' + Buffer.from(svg).toString('base64');
    96→    96→  } catch (err) {
    97→    97→    console.error('[WhatsAppBerryService] Failed to generate QR Data URL:', err.message);
    98→    98→    return null;
    99→    99→  }
   100→   100→}

   328→   328→  setupListeners(qrcodeLib = null) {
   329→   329→    if (!this.client) return;
   330→   330→
   331→   331→    let lastRawQr = null;
   332→   332→
   333→   333→    const handleQrUpdate = async (value) => {
   334→   334→      if (!value) return;
   335→   335→      let rawText = typeof value === 'string' ? value : (value?.value || value?.qr || String(value));
   336→   336→      if (!rawText || rawText === lastRawQr) return;
   337→   337→      lastRawQr = rawText;
   338→   338→
   339→   339→      // If we already have saved credentials and phone number, do not wipe session on transient QR events
   340→   340→      const credsFile1 = path.join(this.sessionPath, 'creds.json');
   341→   341→      const credsFile2 = path.join(this.sessionPath, this.sessionId, 'creds.json');
   342→   342→      if ((fs.existsSync(credsFile1) || fs.existsSync(credsFile2)) && this.phoneNumber) {
   343→   343→        console.log('[WhatsAppBerryService] Preserving existing WhatsApp session credentials; ignoring transient QR event.');
   344→   344→        return;
   345→   345→      }
   346→   346→
   347→   347→      // Format as official WhatsApp Web pairing link if not already prefixed
   348→   348→      if (rawText.startsWith('2@') || rawText.startsWith('1@')) {
   349→   349→        rawText = `https://wa.me/settings/linked_devices#${rawText}`;
   350→   350→      }
   351→   351→
   352→   352→      try {
   353→   353→        const dataUrl = await formatQrDataUrl(rawText, qrcodeLib);
   354→   354→
   355→   355→        this.qrCode = {
   356→   356→          raw: rawText,
   357→   357→          dataUrl
   358→   358→        };
   359→   359→        this.status = 'QR_READY';
   360→   360→        this.phoneNumber = null;
   361→   361→
   362→   362→        console.log('[WhatsAppBerryService] Official WhatsApp Web pairing QR Code generated successfully.');
   363→   363→        this.emit('qr', this.qrCode);
   364→   364→        this.emit('status', this.status);
   365→   365→      } catch (err) {
   366→   366→        console.error(`[WhatsAppBerryService] QR render error: ${err.message}`);
   367→   367→      }
   368→   368→    };
   369→   369→
   370→   370→    this.client.on('auth.qr', ({ value }) => handleQrUpdate(value));
   371→   371→    this.client.on('qr', (qr) => handleQrUpdate(qr));
   372→   372→    this.client.on('auth.link', ({ value }) => handleQrUpdate(value));
   373→   373→
   374→   374→    this.client.on('auth.pairing_code', ({ code, phoneNumber }) => {
   375→   375→      this.pairingCode = code;
   376→   376→      this.pairingPhone = phoneNumber;
   377→   377→      this.status = 'PAIRING_CODE_READY';
   378→   378→      console.log(`[WhatsAppBerryService] WhatsApp 8-digit Pairing Code generated: ${code} for phone ${phoneNumber}`);
   379→   379→      this.emit('pairing_code', { code, phoneNumber });
   380→   380→      this.emit('status', this.status);
   381→   381→    });
   382→   382→
   383→   383→    this.client.on('connection.open', () => {
   384→   384→      this.status = 'CONNECTED';
   385→   385→      this.qrCode = null;
   386→   386→      this.pairingCode = null;
   387→   387→      this.pairingPhone = null;
   388→   388→
   389→   389→      let phone = null;
   390→   390→      try {
   391→   391→        const rawUser = this.client.socket?.sock?.user?.id || this.client.socket?.sock?.user?.jid;
   392→   392→        if (rawUser) {
   393→   393→          const numberPart = rawUser.split(':')[0].split('@')[0];
   394→   394→          phone = this.cleanPhoneNumber(numberPart);
   395→   395→        }
   396→   396→      } catch {}
   397→   397→
   398→   398→      if (phone) {
   399→   399→        this.phoneNumber = phone;
   400→   400→      }
   401→   401→
   402→   402→      // Save metadata with sessionId so it is NEVER lost on restart!
   403→   403→      try {
   404→   404→        fs.writeFileSync(this.metaFile, JSON.stringify({
   405→   405→          status: 'CONNECTED',
   406→   406→          phoneNumber: this.phoneNumber,
   407→   407→          sessionId: this.sessionId,
   408→   408→          platform: 'BerryProtocol v0.1.10',
   409→   409→          updatedAt: new Date().toISOString()
   410→   410→        }, null, 2));
   411→   411→      } catch {}
   412→   412→
   413→   413→      console.log(`[WhatsAppBerryService] WhatsApp BerryProtocol CONNECTED as ${this.phoneNumber || 'Linked Device'}!`);
   414→   414→      this.emit('ready', {
   415→   415→        phoneNumber: this.phoneNumber,
   416→   416→        platform: 'BerryProtocol v0.1.10',
   417→   417→        sessionId: this.sessionId
   418→   418→      });
   419→   419→      this.emit('status', this.status);
   420→   420→    });
   421→   421→
   422→   422→    this.client.on('connection.close', ({ reason, statusCode }) => {
   423→   423→      console.warn(`[WhatsAppBerryService] Connection closed: reason=${reason}, statusCode=${statusCode}`);
   424→   424→      const isManual = reason === 'manual_logout' || statusCode === 401;
   425→   425→
   426→   426→      if (isManual) {
   427→   427→        this.status = 'DISCONNECTED';
   428→   428→        this.phoneNumber = null;
   429→   429→        this.cleanSessionFiles();
   430→   430→        this.emit('disconnected', { reason: 'logged_out' });
   431→   431→        this.emit('status', this.status);
   432→   432→      } else {
   433→   433→        // While waiting for user to scan QR code, DO NOT wipe QR_READY status or destroy active QR code!
   434→   434→        if (this.status === 'QR_READY' && this.qrCode && !this.phoneNumber) {
   435→   435→          console.log('[WhatsAppBerryService] Handshake cycle during QR pairing wait. Preserving active QR on screen.');
   436→   436→          return;
   437→   437→        }
   438→   438→
   439→   439→        this.status = 'DISCONNECTED';
   440→   440→        this.emit('disconnected', { reason, statusCode });
   441→   441→        this.emit('status', this.status);
   442→   442→
   443→   443→        if (this.shouldKeepPairingAlive !== false && this.phoneNumber) {
   444→   444→          clearTimeout(this.reconnectTimer);
   445→   445→          this.reconnectTimer = setTimeout(() => {
   446→   446→            console.log(`[WhatsAppBerryService] Auto-reconnecting transient WhatsApp session (${this.phoneNumber || this.sessionId})...`);
   447→   447→            this.initialize().catch(err => {

   466→   466→          timestamp: message.timestamp || Date.now()
   467→   467→        });
   468→   468→      } catch {}

   540→   540→  /**
   541→   541→   * Generates an 8-digit WhatsApp pairing code for phone number.
   542→   542→   * Completely bypasses WhatsApp Web QR-scan rate limiting.
   543→   543→   * @param {string} phoneNumber 10-digit or full international phone number
   544→   544→   */
   545→   545→  async requestPairingCode(phoneNumber) {
   546→   546→    if (!phoneNumber) {
   547→   547→      throw new Error('Phone number is required for pairing code linking.');
   548→   548→    }
   549→   549→    let clean = String(phoneNumber).replace(/[^0-9]/g, '');
   550→   550→    if (clean.length === 10) clean = '91' + clean;
   551→   551→    else if (clean.length === 11 && clean.startsWith('0')) clean = '91' + clean.slice(1);
   552→   552→
   553→   553→    await this.disconnect({ cleanSession: true });
   554→   554→    this.ensureDirectories();
   555→   555→
   556→   556→    const { BerryProtocol, qrcode } = await loadDependencies();
   557→   557→
   558→   558→    if (!BerryProtocol) {
   559→   559→      if (process.env.NODE_ENV === 'test') {
   560→   560→        const mockCode = `${Math.floor(1000 + Math.random() * 9000)}${Math.floor(1000 + Math.random() * 9000)}`;
   561→   561→        this.pairingCode = mockCode;
   562→   562→        this.pairingPhone = clean;
   563→   563→        this.status = 'PAIRING_CODE_READY';
   564→   564→        return { success: true, pairingCode: mockCode, phoneNumber: clean };
   565→   565→      }
   566→   566→      throw new Error('BerryProtocol engine is not available. Please ensure dependencies are installed.');
   567→   567→    }
   568→   568→
   569→   569→    this.status = 'CONNECTING';
   570→   570→    this.pairingCode = null;
   571→   571→    this.pairingPhone = clean;
   572→   572→
   573→   573→    // Fresh session ID prevents device identity collisions on WhatsApp server
   574→   574→    this.sessionId = `youpi_berry_${Date.now()}`;
   575→   575→    try {
   576→   576→      fs.writeFileSync(this.metaFile, JSON.stringify({
   577→   577→        status: 'CONNECTING',
   578→   578→        phoneNumber: clean,
   579→   579→        sessionId: this.sessionId,
   580→   580→        platform: 'BerryProtocol v0.1.10',
   581→   581→        updatedAt: new Date().toISOString()
   582→   582→      }, null, 2));
   583→   583→    } catch {}
   584→   584→
   585→   585→    this.client = new BerryProtocol({
   586→   586→      sessionId: this.sessionId,
   587→   587→      authFolder: this.sessionPath,
   588→   588→      databasePath: this.databasePath,
   589→   589→      printQrInTerminal: false,
   590→   590→      reconnectDelayMs: 3000,
   591→   591→      reconnectMaxAttempts: 15
   592→   592→    });
   593→   593→
   594→   594→    this.setupListeners(qrcode);
   595→   595→
   596→   596→    // Call connectWithPairingCode on client
   597→   597→    this.client.connectWithPairingCode(clean).catch(err => {
   598→   598→      console.warn(`[WhatsAppBerryService] connectWithPairingCode notice: ${err.message}`);
   599→   599→    });
   600→   600→
   601→   601→    return new Promise((resolve, reject) => {
   602→   602→      const timer = setTimeout(() => {
   603→   603→        if (this.pairingCode) {
   604→   604→          resolve({ success: true, pairingCode: this.pairingCode, phoneNumber: clean });
   605→   605→        } else {
   606→   606→          reject(new Error('Timed out waiting for WhatsApp pairing code from WhatsApp servers.'));
   607→   607→        }
   608→   608→      }, 9000);
   609→   609→
   610→   610→      const onCode = (data) => {
   611→   611→        clearTimeout(timer);
   612→   612→        this.removeListener('pairing_code', onCode);
   613→   613→        resolve({ success: true, pairingCode: data.code, phoneNumber: clean });
   614→   614→      };
   615→   615→
   616→   616→      this.once('pairing_code', onCode);
   617→   617→    });
   618→   618→  }

   620→   620→  /**
   621→   621→   * Dispatches WhatsApp OTP message via BerryProtocol
   622→   622→   * @param {string} to 10-digit or full phone number
   623→   623→   * @param {string} otpCode 6-digit verification code

   626→   626→  async sendOtp(to, otpCode, customerName = '') {
   627→   627→    const greeting = customerName ? `Hello *${customerName}*!` : 'Hello!';
   628→   628→    const message =
   629→   629→      `🔐 *YOUPI PG Verification Code*\n\n` +
   630→   630→      `${greeting} Your one-time login OTP is: *${otpCode}*\n\n` +

   637→   637→  /**
   638→   638→   * Universal message sender via BerryProtocol
   639→   639→   * @param {string} to
   640→   640→   * @param {string} message
   641→   641→   */
   642→   642→  async sendMessage(to, message) {
   643→   643→    const cleanTo = String(to || '').replace(/[^0-9]/g, '');
   644→   644→    const dummyPrefixes = ['910000000000', '0000000000', '919999999999'];
   645→   645→    const isDummy = dummyPrefixes.some(prefix => cleanTo.startsWith(prefix));
   646→   646→
   647→   647→    // Unit test safeguard: If unit test running without forceRealDispatch, simulate dispatch
   648→   648→    if (isDummy || (process.env.NODE_ENV === 'test' && !this.forceRealDispatch)) {
   649→   649→      console.log(`[WhatsAppBerryService] Test guardrail: Simulated OTP message to ${to}`);
   650→   650→      return {
   651→   651→        success: true,
   652→   652→        delivered: true,
   653→   653→        simulated: true,
   654→   654→        mode: 'mock_test_safeguard',
   655→   655→        id: `msg_${Date.now()}`
   656→   656→      };
   657→   657→    }
   658→   658→
   659→   659→    if (this.isSimulatedMode) {
   660→   660→      console.log(`[WhatsAppBerryService] Simulated WhatsApp message to ${to}: "${message.slice(0, 50)}..."`);
   661→   661→      return {
   662→   662→        success: true,
   663→   663→        delivered: true,
   664→   664→        simulated: true,
   665→   665→        id: `sim_berry_${Date.now()}`,
   666→   666→        status: this.status
   667→   667→      };
   668→   668→    }
   669→   669→
   670→   670→    const hasCreds = fs.existsSync(path.join(this.sessionPath, this.sessionId, 'creds.json')) ||
   671→   671→                     fs.existsSync(path.join(this.sessionPath, 'creds.json'));
   672→   672→
   673→   673→    if ((!this.client || this.status !== 'CONNECTED') && hasCreds && !this.isInitializing) {
   674→   674→      console.log(`[WhatsAppBerryService] Reconnecting saved WhatsApp session before message dispatch...`);
   675→   675→      try {
   676→   676→        await this.initialize();
   677→   677→        let attempts = 0;
   678→   678→        while (this.status !== 'CONNECTED' && attempts < 10) {
   679→   679→          await new Promise(r => setTimeout(r, 500));
   680→   680→          attempts++;
   681→   681→        }
   682→   682→      } catch (e) {}
   683→   683→    }
   684→   684→
   685→   685→    if (!this.client || this.status !== 'CONNECTED') {
   686→   686→      console.warn(`[WhatsAppBerryService] WhatsApp BerryProtocol not connected (Status: ${this.status}). Message logged for ${to}.`);
   687→   687→      return {
   688→   688→        success: false,
   689→   689→        delivered: false,
   690→   690→        queued: true,
   691→   691→        status: this.status,
   692→   692→        message: `WhatsApp is not connected (Status: ${this.status}). Please link WhatsApp in Merchant Admin.`
   693→   693→      };
   694→   694→    }
   695→   695→
   696→   696→    const jid = this.formatJid(to);
   697→   697→    if (!jid) {
   698→   698→      return { success: false, delivered: false, error: `Invalid recipient phone number: ${to}` };
   699→   699→    }
   700→   700→
   701→   701→    try {
   702→   702→      let result;
   703→   703→      // BerryProtocol API: sendText(to, text) or sendMessage(to, { text })
   704→   704→      if (typeof this.client.sendText === 'function') {
   705→   705→        result = await this.client.sendText(jid, message);
   706→   706→      } else if (typeof this.client.sendMessage === 'function') {
   707→   707→        result = await this.client.sendMessage(jid, { text: message });
   708→   708→      } else if (this.client.socket && typeof this.client.socket.sendMessage === 'function') {
   709→   709→        result = await this.client.socket.sendMessage(jid, { text: message });
   710→   710→      } else {
   711→   711→        throw new Error('No compatible BerryProtocol message sending method available');
   712→   712→      }
   713→   713→
   714→   714→      console.log(`[WhatsAppBerryService] Outbound WhatsApp message delivered to ${jid} (ID: ${result?.id || 'sent'})`);
   715→   715→      return {
   716→   716→        success: true,
   717→   717→        delivered: true,
   718→   718→        id: result?.id || `berry_${Date.now()}`,
   719→   719→        status: this.status,
   720→   720→        result
   721→   721→      };
   722→   722→    } catch (err) {
   723→   723→      console.error(`[WhatsAppBerryService] Failed to send message to ${to}: ${err.message}`);
   724→   724→      if (err.message && (err.message.includes('Connection Closed') || err.message.includes('closed') || err.message.includes('socket'))) {
   725→   725→        this.status = 'DISCONNECTED';
   726→   726→        this.emit('status', this.status);
   727→   727→        console.log(`[WhatsAppBerryService] Detected broken socket during send. Triggering immediate reconnect...`);
   728→   728→        this.initialize().catch(() => {});
   729→   729→      }
   730→   730→      return { success: false, delivered: false, error: err.message, status: this.status };
   731→   731→    }
   732→   732→  }
