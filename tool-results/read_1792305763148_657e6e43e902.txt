    57→/**
    58→ * Ensures any pairing string or QR data is formatted as an official WhatsApp Web
    59→ * pairing link (https://wa.me/settings/linked_devices#2@...) and converted to an
    60→ * authentic, renderable Data URI (PNG or SVG).
    61→ */
    62→export async function formatQrDataUrl(rawText, qrcodeModule = null) {
    63→  if (!rawText) return null;
    64→  if (typeof rawText === 'string' && rawText.startsWith('data:image/')) {
    65→    return rawText;
    66→  }
    67→
    68→  let text = typeof rawText === 'string' ? rawText : String(rawText);
    69→  if (text.startsWith('2@') || text.startsWith('1@')) {
    70→    text = `https://wa.me/settings/linked_devices#${text}`;
    71→  }
    72→
    73→  let qrEngine = qrcodeModule || qrcodeLib;
    74→  if (!qrEngine) {
    75→    const deps = await loadDependencies();
    76→    qrEngine = deps.qrcode;
    77→  }
    78→  if (qrEngine && typeof qrEngine.toDataURL === 'function') {
    79→    try {
    80→      const dataUrl = await qrEngine.toDataURL(text, {
    81→        width: 320,
    82→        margin: 2,
    83→        errorCorrectionLevel: 'M',
    84→        color: { dark: '#0f172a', light: '#ffffff' }
    85→      });
    86→      if (dataUrl && dataUrl.startsWith('data:image/')) {
    87→        return dataUrl;
    88→      }
    89→    } catch (e) {
    90→      console.warn('[WhatsAppBerryService] qrcode.toDataURL notice:', e.message);
    91→    }
    92→  }
    93→  try {
    94→    const svg = generateQrSvg(text, 320, 'M');
    95→    return 'data:image/svg+xml;base64,' + Buffer.from(svg).toString('base64');
    96→  } catch (err) {
    97→    console.error('[WhatsAppBerryService] Failed to generate QR Data URL:', err.message);
    98→    return null;
    99→  }
   100→}

   341→      const credsFile2 = path.join(this.sessionPath, this.sessionId, 'creds.json');
   342→      if ((fs.existsSync(credsFile1) || fs.existsSync(credsFile2)) && this.phoneNumber) {

   344→        return;

   348→      if (rawText.startsWith('2@') || rawText.startsWith('1@')) {

   540→  /**
   541→   * Generates an 8-digit WhatsApp pairing code for phone number.
   542→   * Completely bypasses WhatsApp Web QR-scan rate limiting.
   543→   * @param {string} phoneNumber 10-digit or full international phone number
   544→   */
   545→  async requestPairingCode(phoneNumber) {
   546→    if (!phoneNumber) {
   547→      throw new Error('Phone number is required for pairing code linking.');
   548→    }
   549→    let clean = String(phoneNumber).replace(/[^0-9]/g, '');
   550→    if (clean.length === 10) clean = '91' + clean;
   551→    else if (clean.length === 11 && clean.startsWith('0')) clean = '91' + clean.slice(1);
   552→
   553→    await this.disconnect({ cleanSession: true });
   554→    this.ensureDirectories();
   555→
   556→    const { BerryProtocol, qrcode } = await loadDependencies();
   557→
   558→    if (!BerryProtocol) {
   559→      if (process.env.NODE_ENV === 'test') {
   560→        const mockCode = `${Math.floor(1000 + Math.random() * 9000)}${Math.floor(1000 + Math.random() * 9000)}`;
   561→        this.pairingCode = mockCode;
   562→        this.pairingPhone = clean;
   563→        this.status = 'PAIRING_CODE_READY';
   564→        return { success: true, pairingCode: mockCode, phoneNumber: clean };
   565→      }
   566→      throw new Error('BerryProtocol engine is not available. Please ensure dependencies are installed.');
   567→    }
   568→
   569→    this.status = 'CONNECTING';
   570→    this.pairingCode = null;
   571→    this.pairingPhone = clean;
   572→
   573→    // Fresh session ID prevents device identity collisions on WhatsApp server
   574→    this.sessionId = `youpi_berry_${Date.now()}`;
   575→    try {
   576→      fs.writeFileSync(this.metaFile, JSON.stringify({
   577→        status: 'CONNECTING',
   578→        phoneNumber: clean,
   579→        sessionId: this.sessionId,
   580→        platform: 'BerryProtocol v0.1.10',
   581→        updatedAt: new Date().toISOString()
   582→      }, null, 2));
   583→    } catch {}
   584→
   585→    this.client = new BerryProtocol({
   586→      sessionId: this.sessionId,
   587→      authFolder: this.sessionPath,
   588→      databasePath: this.databasePath,
   589→      printQrInTerminal: false,
   590→      reconnectDelayMs: 3000,
   591→      reconnectMaxAttempts: 15
   592→    });
   593→
   594→    this.setupListeners(qrcode);
   595→
   596→    // Call connectWithPairingCode on client
   597→    this.client.connectWithPairingCode(clean).catch(err => {
   598→      console.warn(`[WhatsAppBerryService] connectWithPairingCode notice: ${err.message}`);
   599→    });
   600→
   601→    return new Promise((resolve, reject) => {
   602→      const timer = setTimeout(() => {
   603→        if (this.pairingCode) {
   604→          resolve({ success: true, pairingCode: this.pairingCode, phoneNumber: clean });
   605→        } else {
   606→          reject(new Error('Timed out waiting for WhatsApp pairing code from WhatsApp servers.'));
   607→        }
   608→      }, 9000);
   609→
   610→      const onCode = (data) => {
   611→        clearTimeout(timer);
   612→        this.removeListener('pairing_code', onCode);
   613→        resolve({ success: true, pairingCode: data.code, phoneNumber: clean });
   614→      };
   615→
   616→      this.once('pairing_code', onCode);
   617→    });
   618→  }
   619→
   620→  /**
   621→   * Dispatches WhatsApp OTP message via BerryProtocol
   622→   * @param {string} to 10-digit or full phone number
   623→   * @param {string} otpCode 6-digit verification code
   624→   * @param {string} [customerName]
   625→   */
   626→  async sendOtp(to, otpCode, customerName = '') {
   627→    const greeting = customerName ? `Hello *${customerName}*!` : 'Hello!';
   628→    const message =
   629→      `🔐 *YOUPI PG Verification Code*\n\n` +
   630→      `${greeting} Your one-time login OTP is: *${otpCode}*\n\n` +
   631→      `⏱ _Valid for 5 minutes. Do not share this code with anyone._\n\n` +
   632→      `— *YOUPI PG Secure Payments*`;
   633→
   634→    return this.sendMessage(to, message);
   635→  }
   636→
   637→  /**
   638→   * Universal message sender via BerryProtocol
   639→   * @param {string} to
   640→   * @param {string} message
   641→   */
   642→  async sendMessage(to, message) {
   643→    const cleanTo = String(to || '').replace(/[^0-9]/g, '');
   644→    const dummyPrefixes = ['910000000000', '0000000000', '919999999999'];
   645→    const isDummy = dummyPrefixes.some(prefix => cleanTo.startsWith(prefix));
   646→
   647→    // Unit test safeguard: If unit test running without forceRealDispatch, simulate dispatch
   648→    if (isDummy || (process.env.NODE_ENV === 'test' && !this.forceRealDispatch)) {
   649→      console.log(`[WhatsAppBerryService] Test guardrail: Simulated OTP message to ${to}`);
   650→      return {
   651→        success: true,
   652→        delivered: true,
   653→        simulated: true,
   654→        mode: 'mock_test_safeguard',
   655→        id: `msg_${Date.now()}`
   656→      };
   657→    }
   658→
   659→    if (this.isSimulatedMode) {
   660→      console.log(`[WhatsAppBerryService] Simulated WhatsApp message to ${to}: "${message.slice(0, 50)}..."`);
   661→      return {
   662→        success: true,
   663→        delivered: true,
   664→        simulated: true,
   665→        id: `sim_berry_${Date.now()}`,
   666→        status: this.status
   667→      };
   668→    }
   669→
   670→    const hasCreds = fs.existsSync(path.join(this.sessionPath, this.sessionId, 'creds.json')) ||
   671→                     fs.existsSync(path.join(this.sessionPath, 'creds.json'));
   672→
   673→    if ((!this.client || this.status !== 'CONNECTED') && hasCreds && !this.isInitializing) {
   674→      console.log(`[WhatsAppBerryService] Reconnecting saved WhatsApp session before message dispatch...`);
   675→      try {
   676→        await this.initialize();
   677→        let attempts = 0;
   678→        while (this.status !== 'CONNECTED' && attempts < 10) {
   679→          await new Promise(r => setTimeout(r, 500));
   680→          attempts++;
   681→        }
   682→      } catch (e) {}
   683→    }
   684→
   685→    if (!this.client || this.status !== 'CONNECTED') {
   686→      console.warn(`[WhatsAppBerryService] WhatsApp BerryProtocol not connected (Status: ${this.status}). Message logged for ${to}.`);
   687→      return {
   688→        success: false,
   689→        delivered: false,
   690→        queued: true,
   691→        status: this.status,
   692→        message: `WhatsApp is not connected (Status: ${this.status}). Please link WhatsApp in Merchant Admin.`
   693→      };
   694→    }
   695→
   696→    const jid = this.formatJid(to);
   697→    if (!jid) {
   698→      return { success: false, delivered: false, error: `Invalid recipient phone number: ${to}` };
   699→    }
   700→
   701→    try {
   702→      let result;
   703→      // BerryProtocol API: sendText(to, text) or sendMessage(to, { text })
   704→      if (typeof this.client.sendText === 'function') {
   705→        result = await this.client.sendText(jid, message);
   706→      } else if (typeof this.client.sendMessage === 'function') {
   707→        result = await this.client.sendMessage(jid, { text: message });
   708→      } else if (this.client.socket && typeof this.client.socket.sendMessage === 'function') {
   709→        result = await this.client.socket.sendMessage(jid, { text: message });
   710→      } else {
   711→        throw new Error('No compatible BerryProtocol message sending method available');
   712→      }
   713→
   714→      console.log(`[WhatsAppBerryService] Outbound WhatsApp message delivered to ${jid} (ID: ${result?.id || 'sent'})`);
   715→      return {
   716→        success: true,
   717→        delivered: true,
   718→        id: result?.id || `berry_${Date.now()}`,
   719→        status: this.status,
   720→        result
   721→      };
   722→    } catch (err) {
   723→      console.error(`[WhatsAppBerryService] Failed to send message to ${to}: ${err.message}`);
   724→      if (err.message && (err.message.includes('Connection Closed') || err.message.includes('closed') || err.message.includes('socket'))) {
   725→        this.status = 'DISCONNECTED';
   726→        this.emit('status', this.status);
   727→        console.log(`[WhatsAppBerryService] Detected broken socket during send. Triggering immediate reconnect...`);
   728→        this.initialize().catch(() => {});
   729→      }
   730→      return { success: false, delivered: false, error: err.message, status: this.status };
   731→    }
   732→  }
   733→}
