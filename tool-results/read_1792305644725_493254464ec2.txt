     5→     5→/**
     6→     6→ * Validates device access to an active payment session.
     7→     7→ * Enforces:
     8→     8→ * 1. If session has no device binding yet, binds to requesting device if provided.
     9→     9→ * 2. If session is bound to Device A, Device B receives DUPLICATE_SESSION_DETECTED (HTTP 409).
    10→    10→ */
    11→    11→function checkSessionDeviceAccess(session, req, body = null, ctx = null) {
    12→    12→  // If session is already in a terminal state, allow access so terminal status can be rendered
    13→    13→  if ([PaymentStatus.SUCCESS, PaymentStatus.FAILED, PaymentStatus.EXPIRED, PaymentStatus.CANCELLED].includes(session.status)) {
    14→    14→    return { allowed: true, boundDeviceId: session.deviceId, isTerminal: true };
    15→    15→  }
    16→    16→
    17→    17→  // Internal server-side HTML render calls bypass device blocking
    18→    18→  if (req.headers['x-internal-render'] === 'true') {
    19→    19→    return { allowed: true, boundDeviceId: session.deviceId };
    20→    20→  }
    21→    21→
    22→    22→  const requestingDeviceId = req.headers['x-device-id'] ||
    23→    23→    body?.deviceId ||
    24→    24→    (req.query && req.query.deviceId) ||
    25→    25→    null;
    26→    26→
    27→    27→  // If session does not have a device binding yet:
    28→    28→  if (!session.deviceId) {
    29→    29→    if (requestingDeviceId) {
    30→    30→      if (ctx?.deviceRepo) {
    31→    31→        let dev = ctx.deviceRepo.findById(requestingDeviceId);
    32→    32→        if (!dev) {
    33→    33→          try {
    34→    34→            ctx.deviceRepo.create({
    35→    35→              id: requestingDeviceId,
    36→    36→              publicKey: `adhoc_${requestingDeviceId}`,
    37→    37→              fingerprintHash: `fp_${requestingDeviceId}`
    38→    38→            });
    39→    39→          } catch {}
    40→    40→        } else {
    41→    41→          ctx.deviceRepo.touch(requestingDeviceId);
    42→    42→        }
    43→    43→      }
    44→    44→      if (ctx?.sessionRepo) {
    45→    45→        try {
    46→    46→          ctx.sessionRepo.bindDevice(session.id, requestingDeviceId);
    47→    47→          session.deviceId = requestingDeviceId;
    48→    48→        } catch (err) {
    49→    49→          console.warn('[checkSessionDeviceAccess] bindDevice error:', err.message);
    50→    50→        }
    51→    51→      }
    52→    52→    }
    53→    53→    return { allowed: true, boundDeviceId: session.deviceId };
    54→    54→  }
    55→    55→
    56→    56→  // If session HAS a bound device:
    57→    57→  // Allow if requesting device matches bound device OR if requesting device is omitted on same session
    58→    58→  if (!requestingDeviceId || requestingDeviceId === session.deviceId) {
    59→    59→    if (requestingDeviceId && ctx?.deviceRepo) {
    60→    60→      ctx.deviceRepo.touch(requestingDeviceId);
    61→    61→    }
    62→    62→    return { allowed: true, boundDeviceId: session.deviceId };
    63→    63→  }
    64→    64→
    65→    65→  // Device mismatch: only reject if an explicitly DIFFERENT device ID was provided!
    66→    66→  return {
    67→    67→    allowed: false,
    68→    68→    statusCode: 409,
    69→    69→    body: {
    70→    70→      error: 'DUPLICATE_SESSION_DETECTED',
    71→    71→      session_id: session.id,
    72→    72→      conflicting_session_id: session.id,
    73→    73→      action: 'END_SESSION_AND_CREATE_NEW',
    74→    74→      message: 'This payment session was created on another device. For security reasons, this session cannot be continued from this device.',
    75→    75→      boundDeviceId: session.deviceId,
    76→    76→      requestingDeviceId
    77→    77→    }
    78→    78→  };
    79→    79→}

   135→   135→          falseNarrativeRemainingSeconds: remainingSeconds

   138→   138→        return {
   139→   139→          status: 'ISSUES',
   140→   140→          tier: 'RED',
   141→   141→          label: 'Server Facing Issues',

   148→   148→          isFalseNarrative: true,

   184→   184→        if (data.isPaired === true && data.status === 'CONNECTED' && data.mode === 'BACKGROUND' && (data.isPolling === true || isRecent)) {

   207→   207→          if (data.status === 'UP') {
   208→   208→            pollerActive = true;

   216→   216→          if (res.ok) {

   238→   238→        gmailHealthy = true;

   242→   242→      }
   243→   243→    } else {
   244→   244→      // Check if browser-based Gmail automator is actively logged in

   254→   254→        }
   255→   255→      } catch (e) {}
   256→   256→
   257→   257→      if (!gmailHealthy) {
   258→   258→        gmailReason = 'Gmail OAuth not configured';

   271→   271→    : enabledMethods.includes('CASH_COUNTER');

   273→   273→  if (smsHealthy && gmailHealthy) {
   274→   274→    return {
   275→   275→      status: 'WORKING',

   277→   277→      label: 'Server Is Working',
   278→   278→      isWorking: true,
   279→   279→      reason: 'All observers active: Gmail OAuth polling & SMS Gateway connected',
   280→   280→      smsHealthy: true,
   281→   281→      gmailHealthy: true,
   282→   282→      warning: null,
   283→   283→      recommendedAlternative: null
   284→   284→    };
   285→   285→  }
   286→   286→
   287→   287→  if (smsHealthy || gmailHealthy) {
   288→   288→    return {
   289→   289→      status: 'BUSY',
   290→   290→      tier: 'YELLOW',
   291→   291→      label: 'UPI Server Busy',
   292→   292→      isWorking: true,
   293→   293→      reason: smsHealthy ? 'SMS observer active, Gmail observer offline' : 'Gmail observer active, SMS observer offline',
   294→   294→      smsHealthy,
   295→   295→      gmailHealthy,
   296→   296→      warning: 'Your payment may go through extra processing seconds.',
   297→   297→      recommendedAlternative: null
   298→   298→    };
   299→   299→  }
   300→   300→
   301→   301→  return {

   308→   308→    gmailHealthy: false,

   311→   311→  };
   312→   312→}

   314→   314→export function registerCheckoutRoutes(server, ctx) {
   315→   315→  // GET /api/v1/checkout/:id/state
   316→   316→  server.get('/api/v1/checkout/:id/state', async (req, res, params) => {
   317→   317→    const session = ctx.sessionRepo.findById(params.id);
   318→   318→    if (!session) {
   319→   319→      HttpServer.json(res, 404, { error: `Checkout session not found: ${params.id}` });
   320→   320→      return;
   321→   321→    }
   322→   322→
   323→   323→    // Server-side Ban & Fraud Enforcement
   324→   324→    const isCustomerBanned = session.customerId && ctx.customerRepo && ctx.customerRepo.isBanned(session.customerId);
   325→   325→    const isSessionBlocked = session.status === 'BLOCKED' || (session.metadata?.banReason === 'FRAUDULENT_PAYMENT_INJECTION');
   326→   326→
   327→   327→    if (isCustomerBanned || isSessionBlocked) {
   328→   328→      const cust = session.customerId ? ctx.customerRepo.findById(session.customerId) : null;
   329→   329→      HttpServer.json(res, 200, {
   330→   330→        id: session.id,
   331→   331→        orderId: session.orderId,
   332→   332→        amount: toRupeesString(session.amountPaise),
   333→   333→        status: 'BLOCKED',
   334→   334→        isBanned: true,
   335→   335→        fraudDetected: true,
   336→   336→        customerName: cust?.name || 'Customer',
   337→   337→        blockedNotice: {
   338→   338→          title: 'FRAUDULENT ACTIVITY DETECTED',
   339→   339→          subtitle: 'Payment Session Terminated & Account Suspended',
   340→   340→          message: 'Our automated verification engine detected fraudulent payment activity associated with this account. Access has been immediately suspended while our security operations team investigates the issue.',
   341→   341→          investigating: true,
   342→   342→          reason: cust?.banReason || 'FRAUDULENT_PAYMENT_INJECTION'
   343→   343→        }
   344→   344→      });
   345→   345→      return;
   346→   346→    }
   347→   347→
   348→   348→    // Authoritative Server-Side Expiration Enforcer
   349→   349→    const now = Date.now();
   350→   350→    if (session.status === PaymentStatus.ACTIVE && session.expiresAt && session.expiresAt.getTime() <= now) {
   351→   351→      ctx.sessionRepo.transitionToTerminal(session.id, PaymentStatus.EXPIRED);
   352→   352→      session.status = PaymentStatus.EXPIRED;
   353→   353→      if (session.receivingAccountId) {
   354→   354→        ctx.queueEngine.promoteNext(session.receivingAccountId);
   355→   355→      }
   356→   356→      server.broadcastSse(session.id, 'status_change', { status: PaymentStatus.EXPIRED });
   357→   357→    } else if (session.status === PaymentStatus.PROCESSING && session.processingExpiresAt && session.processingExpiresAt.getTime() <= now) {
   358→   358→      ctx.sessionRepo.transitionToTerminal(session.id, PaymentStatus.FAILED);
   359→   359→      session.status = PaymentStatus.FAILED;
   360→   360→      if (session.receivingAccountId) {
   361→   361→        ctx.queueEngine.promoteNext(session.receivingAccountId);
   362→   362→      }
   363→   363→      server.broadcastSse(session.id, 'status_change', { status: PaymentStatus.FAILED });
   364→   364→    }
   365→   365→
   366→   366→    // Terminal session handling: if session is terminal, return terminal state directly (CASE 4)
   367→   367→    if ([PaymentStatus.SUCCESS, PaymentStatus.FAILED, PaymentStatus.EXPIRED, PaymentStatus.CANCELLED].includes(session.status)) {
   368→   368→      const merchant = ctx.merchantRepo.findById(session.merchantId);
   369→   369→      const branding = {
   370→   370→        appName: 'YOUPI PG',
   371→   371→        primaryColor: '#2563eb',
   372→   372→        ...(merchant?.branding || {})
   373→   373→      };
   374→   374→      HttpServer.json(res, 200, {
   375→   375→        id: session.id,
   376→   376→        paymentId: session.id,
   377→   377→        orderId: session.orderId,
   378→   378→        amount: toRupeesString(session.amountPaise),
   379→   379→        amountPaise: session.amountPaise.toString(),
   380→   380→        currency: session.currency,
   381→   381→        status: session.status,
   382→   382→        timeRemainingSeconds: 0,
   383→   383→        isTerminal: true,
   384→   384→        branding
   385→   385→      });
   386→   386→      return;
   387→   387→    }
   388→   388→
   389→   389→    // Device-level session binding validation (Section 8 & 9)
   390→   390→    const deviceAccess = checkSessionDeviceAccess(session, req, null, ctx);
   391→   391→    if (!deviceAccess.allowed) {
   392→   392→      HttpServer.json(res, deviceAccess.statusCode || 409, deviceAccess.body);
   393→   393→      return;
   394→   394→    }
   395→   395→
   396→   396→    const merchant = ctx.merchantRepo.findById(session.merchantId);
   397→   397→    const snapshot = session.metadata?.receivingAccountSnapshot;
   398→   398→    let receivingAccount = session.receivingAccountId
   399→   399→      ? ctx.accountRepo.findById(session.receivingAccountId)
   400→   400→      : null;
   401→   401→
   402→   402→    if (!receivingAccount && snapshot?.id) {
   403→   403→      receivingAccount = ctx.accountRepo.findById(snapshot.id);
   404→   404→    }
   405→   405→    if (!receivingAccount && session.method === 'UPI') {
   406→   406→      receivingAccount = ctx.accountRepo.findActiveForMerchant(session.merchantId);
   407→   407→    }
   408→   408→
   409→   409→    const payeeVpa = receivingAccount?.upiId || snapshot?.upiId || null;
   410→   410→    const payeeName = receivingAccount?.payeeName || snapshot?.payeeName || merchant?.businessName || 'Merchant Store';
   411→   411→    const isMerchantEnabled = receivingAccount
   412→   412→      ? Boolean(receivingAccount.isMerchantEnabled)
   413→   413→      : (snapshot?.isMerchantEnabled !== undefined ? Boolean(snapshot.isMerchantEnabled) : true);
   414→   414→    const upiMode = isMerchantEnabled ? 'MERCHANT' : 'NON_MERCHANT';
   415→   415→
   416→   416→    let timeRemainingSeconds = 0;
   417→   417→    if (session.status === PaymentStatus.ACTIVE && session.expiresAt) {
   418→   418→      timeRemainingSeconds = Math.max(0, Math.ceil((session.expiresAt.getTime() - now) / 1000));
   419→   419→    } else if (session.status === PaymentStatus.PROCESSING && session.processingExpiresAt) {
   420→   420→      timeRemainingSeconds = Math.max(0, Math.ceil((session.processingExpiresAt.getTime() - now) / 1000));
   421→   421→    } else if (session.status === PaymentStatus.QUEUED) {
   422→   422→      timeRemainingSeconds = ctx.queueEngine.calculateEstimatedWaitSeconds(
   423→   423→        session.receivingAccountId,
   424→   424→        session.queuePosition
   425→   425→      );
   426→   426→    }
   427→   427→
   428→   428→    let upiUri = null;
   429→   429→    if (session.status === PaymentStatus.ACTIVE && payeeVpa) {
   430→   430→      try {
   431→   431→        upiUri = buildUpiUri({
   432→   432→          payeeVpa,
   433→   433→          payeeName,
   434→   434→          amountPaise: session.amountPaise,
   435→   435→          transactionRef: session.orderId,
   436→   436→          currency: session.currency || 'INR'
   437→   437→        });
   438→   438→      } catch {
   439→   439→        upiUri = null;
   440→   440→      }
   441→   441→    }
   442→   442→
